## 1. Struktur Direktori & Penempatan File
Pastikan pengembangan mengikuti arsitektur ini secara ketat[cite: 21]:
*   **API Endpoints**: Seluruh Route Handlers harus ditempatkan di dalam folder `/app/api/`.
*   **Logika Bisnis & AI**: Seluruh fungsi murni dan validasi harus berada di dalam folder `/lib/` (misalnya: `scoring.ts`, `status-machine.ts`, `overlap.ts`, `ai/summarizer.ts`, dan skema validasi Zod). Jangan meletakkan logika bisnis di dalam komponen UI atau Route Handlers[cite: 37].
*   **Database & Seed**: Konfigurasi skema dan data awal harus diletakkan di dalam folder `/prisma/` (khususnya `schema.prisma` dan `seed.ts`).

## 2. Aturan Khusus dari PRD
*   **Tipe Data Uang**: Semua nilai uang (pagu, penawaran, hargaSatuan) **WAJIB** disimpan sebagai `BigInt`.
*   **Keamanan NIK**: NIK tidak pernah disimpan mentah, melainkan menggunakan `nikHash` yang diindeks[cite: 31, 57].
*   **Ekstensi Database**: Aplikasi membutuhkan pencarian kemiripan teks, sehingga ekstensi PostgreSQL `pg_trgm` wajib diaktifkan[cite: 51].
*   **Integritas Relasi**: Untuk memenuhi cakupan T1 (Desa, Proyek, RabItem, Kelompok, AnggotaKelompok, Proposal, RingkasanAi, Skor, FlagKemiripan) tanpa galat validasi Prisma, model pendukung seperti `Pengguna` harus disertakan dalam bentuk *stub* (minimalist) agar `foreign key` seperti `ketuaId` dan `penggunaId` valid[cite: 56, 57].

## 3. Draf `schema.prisma` (Subset T1)
Terapkan kode berikut ke dalam file `prisma/schema.prisma`. 

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")
}

// ================= ENUMERASI ================= 
enum KelasProyek { KECIL \n MENENGAH \n BESAR }
enum Jalur { SAYEMBARA \n VENDOR }
enum StatusProyek { PEMATANGAN \n TENDER_DIBUKA \n PEMILIHAN_VENDOR \n SELEKSI \n VERIFIKASI_KOMPETENSI \n MASA_SANGGAH \n PENGERJAAN \n SELESAI \n DIBATALKAN }
enum StatusAnalisis { MENUNGGU \n BERHASIL \n GAGAL }
enum JenisFlag { TEKS \n ANGGOTA }
enum KeputusanFlag { BELUM \n WAJAR \n KLARIFIKASI \n TIDAK_SAH }

// ================= STUB PENDUKUNG ================= 
// Disediakan agar relasi pada Kelompok dan AnggotaKelompok tervalidasi
model Pengguna {
  id                String            @id @default(cuid())
  nama              String
  kelompokDipimpin  Kelompok[]
  anggotaKelompok   AnggotaKelompok[]
}

// ================= MODEL INTI T1 ================= 

model Desa {
  id          String   @id @default(cuid())
  nama        String
  kecamatan   String
  kabupaten   String
  tahun       Int      @default(0)
  pagu        BigInt
  pagukdmp    BigInt
  konfigurasi Json?
  createdAt   DateTime @default(now())
  proyek      Proyek[]
}

model Proyek {
  id               String        @id @default(cuid())
  desaId           String
  judul            String
  deskripsi        String
  lokasi           String
  pagu             BigInt
  kelas            KelasProyek
  jalur            Jalur?
  alasanJalur      String?
  status           StatusProyek  @default(PEMATANGAN)
  tenggatProposal  DateTime?
  akhirMasaSanggah DateTime?
  syaratLokalMin   Int           @default(70)
  durasiMin        Int           @default(30)
  durasiMax        Int           @default(90)
  createdAt        DateTime      @default(now())
  
  desa             Desa          @relation(fields: [desaId], references: [id])
  rab              RabItem[]
  proposal         Proposal[]
  flag             FlagKemiripan[]

  @@index([desaId, status])
}

model RabItem {
  id          String  @id @default(cuid())
  proyekId    String
  uraian      String
  volume      Float
  satuan      String
  hargaSatuan BigInt
  
  proyek      Proyek  @relation(fields: [proyekId], references: [id])
  
  @@index([proyekId])
}

model Kelompok {
  id        String   @id @default(cuid())
  nama      String
  ketuaId   String
  createdAt DateTime @default(now())
  
  ketua     Pengguna @relation(fields: [ketuaId], references: [id])
  anggota   AnggotaKelompok[]
  proposal  Proposal[]
}

model AnggotaKelompok {
  id         String   @id @default(cuid())
  kelompokId String
  penggunaId String?
  nama       String?
  rtRw       String?
  nikHash    String?
  ketua      Boolean  @default(false)
  
  kelompok   Kelompok @relation(fields: [kelompokId], references: [id])
  pengguna   Pengguna? @relation(fields: [penggunaId], references: [id])
  
  @@index([nikHash])
  @@index([kelompokId])
}

model Proposal {
  id             String         @id @default(cuid())
  proyekId       String
  kelompokId     String
  penawaran      BigInt
  durasiHari     Int
  tenagaTotal    Int
  tenagaLokal    Int
  narasiMetode   String
  narasiJadwal   String
  narasiRisiko   String
  dokumen        String[]
  kelengkapan    Json
  dikirimPada    DateTime       @default(now())
  statusAnalisis StatusAnalisis @default(MENUNGGU)
  
  proyek         Proyek         @relation(fields: [proyekId], references: [id])
  kelompok       Kelompok       @relation(fields: [kelompokId], references: [id])
  ringkasan      RingkasanAi?
  skor           Skor?
  flagA          FlagKemiripan[] @relation("FlagA")
  flagB          FlagKemiripan[] @relation("FlagB")
  
  @@unique([proyekId, kelompokId])
  @@index([proyekId])
}

model RingkasanAi {
  id          String   @id @default(cuid())
  proposalId  String   @unique
  json        Json
  model       String
  versiPrompt String
  dibuatPada  DateTime @default(now())
  
  proposal    Proposal @relation(fields: [proposalId], references: [id])
}

model Skor {
  id          String   @id @default(cuid())
  proposalId  String   @unique
  total       Int
  rincian     Json
  flags       Json
  versiBobot  String
  dibuatPada  DateTime @default(now())
  
  proposal    Proposal @relation(fields: [proposalId], references: [id])
}

model FlagKemiripan {
  id            String        @id @default(cuid())
  proyekId      String
  proposalAId   String
  proposalBId   String
  nilai         Float
  jenis         JenisFlag
  keputusan     KeputusanFlag @default(BELUM)
  catatan       String?
  
  proyek        Proyek        @relation(fields: [proyekId], references: [id])
  proposalA     Proposal      @relation("FlagA", fields: [proposalAId], references: [id])
  proposalB     Proposal      @relation("FlagB", fields: [proposalBId], references: [id])
  
  @@index([proyekId])
}
